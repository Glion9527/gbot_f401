#ifndef HOST_LINK_V2_USART_H
#define HOST_LINK_V2_USART_H

#include "main.h"

/* 主机仿真的 USART2：IT 发送把 TELEM 复制到测试缓冲区，接收函数只记录调用。 */
typedef struct { uint32_t unused; } USART_TypeDef;
extern USART_TypeDef host_usart2;
#define USART2 (&host_usart2)

typedef struct {
    USART_TypeDef *Instance;
    uint32_t ErrorCode;
} UART_HandleTypeDef;

extern UART_HandleTypeDef huart2;

HAL_StatusTypeDef HAL_UART_Transmit_IT(UART_HandleTypeDef *h, uint8_t *data,
                                       uint16_t size);
HAL_StatusTypeDef HAL_UART_Receive_IT(UART_HandleTypeDef *h, uint8_t *data,
                                      uint16_t size);
HAL_StatusTypeDef HAL_UART_AbortReceive_IT(UART_HandleTypeDef *h);

#define HAL_UART_ERROR_NONE 0U
#define HAL_UART_ERROR_ORE 8U

#endif /* HOST_LINK_V2_USART_H */
